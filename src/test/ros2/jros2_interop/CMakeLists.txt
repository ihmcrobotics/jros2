cmake_minimum_required(VERSION 3.8)
project(jros2_interop)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(example_interfaces REQUIRED)
find_package(rclcpp_action REQUIRED)
# Foxy has neither <pkg>_TARGETS nor timeouts in SyncParametersClient
if("$ENV{ROS_DISTRO}" STREQUAL "foxy")
  add_compile_definitions(JROS2_INTEROP_FOXY)
endif()

function(jros2_interop_executable name source)
  add_executable(${name} ${source})
  if(DEFINED example_interfaces_TARGETS)
    target_link_libraries(${name} rclcpp::rclcpp rclcpp_action::rclcpp_action ${example_interfaces_TARGETS})
  else()
    ament_target_dependencies(${name} rclcpp rclcpp_action example_interfaces)
  endif()
endfunction()

jros2_interop_executable(add_two_ints_server src/server.cpp)
jros2_interop_executable(add_two_ints_client src/client.cpp)
jros2_interop_executable(param_client src/param_client.cpp)
jros2_interop_executable(fibonacci_server src/fibonacci_server.cpp)
jros2_interop_executable(fibonacci_client src/fibonacci_client.cpp)
install(TARGETS add_two_ints_server add_two_ints_client param_client fibonacci_server fibonacci_client DESTINATION lib/${PROJECT_NAME})
ament_package()
