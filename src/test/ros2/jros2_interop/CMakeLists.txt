cmake_minimum_required(VERSION 3.8)
project(jros2_interop)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(example_interfaces REQUIRED)
add_executable(add_two_ints_server src/server.cpp)
target_link_libraries(add_two_ints_server rclcpp::rclcpp ${example_interfaces_TARGETS})
add_executable(add_two_ints_client src/client.cpp)
target_link_libraries(add_two_ints_client rclcpp::rclcpp ${example_interfaces_TARGETS})
install(TARGETS add_two_ints_server add_two_ints_client DESTINATION lib/${PROJECT_NAME})
ament_package()
