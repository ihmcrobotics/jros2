cmake_minimum_required(VERSION 3.8)
project(jros2_interop)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(example_interfaces REQUIRED)
find_package(rclcpp_action REQUIRED)
add_executable(add_two_ints_server src/server.cpp)
target_link_libraries(add_two_ints_server rclcpp::rclcpp ${example_interfaces_TARGETS})
add_executable(add_two_ints_client src/client.cpp)
target_link_libraries(add_two_ints_client rclcpp::rclcpp ${example_interfaces_TARGETS})
add_executable(param_client src/param_client.cpp)
target_link_libraries(param_client rclcpp::rclcpp)
add_executable(fibonacci_server src/fibonacci_server.cpp)
target_link_libraries(fibonacci_server rclcpp::rclcpp rclcpp_action::rclcpp_action ${example_interfaces_TARGETS})
add_executable(fibonacci_client src/fibonacci_client.cpp)
target_link_libraries(fibonacci_client rclcpp::rclcpp rclcpp_action::rclcpp_action ${example_interfaces_TARGETS})
install(TARGETS add_two_ints_server add_two_ints_client param_client fibonacci_server fibonacci_client DESTINATION lib/${PROJECT_NAME})
ament_package()
